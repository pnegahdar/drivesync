package dsreview2

import (
	"database/sql"
	"os"
	"path/filepath"
	"testing"

	ds "github.com/pnegahdar/drivesync"
	_ "modernc.org/sqlite"
)

// A pre-upgrade cross-tenant folder (no allocation) rejects even deletes of
// existing rows, and a replica with a pending local delete then never pulls.
func TestLegacyFolderBlocksDeletesAndDownloads(t *testing.T) {
	dbPath := filepath.Join(t.TempDir(), "meta.sqlite")
	m, _ := ds.OpenSQLiteMetaStore(dbPath)
	defer m.Close()
	s := ds.NewServer(m, ds.NewMemoryBlobStore())
	alice := ds.Principal{Tenant: "acme", Subject: "alice"}
	bob := ds.Principal{Tenant: "partner", Subject: "bob"}
	ac := s.Client(alice)
	f, k := mkFolder(t, ac, ds.Limits{MaxTotalBytes: 1 << 20})
	putFile(t, ac, f, k, "old.txt", 0, []byte("old"))
	if e := ac.Grant(bg, f.ID, bob, ds.Writer); e != nil {
		t.Fatal(e)
	}
	r, dir := attach(t, ac, f, k, "alice-laptop")
	if e := r.Sync(bg); e != nil {
		t.Fatal(e)
	}
	// Simulate a folder created by e8d8ed2: cross-tenant grant, no allocation.
	db, _ := sql.Open("sqlite", dbPath)
	if _, e := db.Exec(`UPDATE folders SET data=json_set(data,'$.Allocated',json('false')) WHERE id=?`, f.ID); e != nil {
		t.Fatal(e)
	}
	db.Close()
	putFileNoFail := func() error {
		pid, _ := ds.PathID(k, f.ID, "x")
		_, e := ac.Commit(bg, f.ID, []ds.Mutation{{PathID: pid, Deleted: true}})
		return e
	}
	_ = putFileNoFail
	// Pure delete of an existing row by the primary owner:
	d, _ := ac.Changes(bg, f.ID, 0)
	_, e := ac.Commit(bg, f.ID, []ds.Mutation{{PathID: d.Rows[0].PathID, BaseVersion: d.Rows[0].Version, Deleted: true}})
	t.Logf("owner deleting an existing file in legacy folder: %v", e)
	// Replica: user deletes a file locally; a teammate's later file never arrives.
	os.Remove(filepath.Join(dir, "old.txt"))
	db2, _ := sql.Open("sqlite", dbPath)
	db2.Exec(`UPDATE folders SET data=json_set(data,'$.Allocated',json('true')) WHERE id=?`, f.ID)
	putFile(t, ac, f, k, "teammate-new.txt", 0, []byte("new"))
	db2.Exec(`UPDATE folders SET data=json_set(data,'$.Allocated',json('false')) WHERE id=?`, f.ID)
	db2.Close()
	se := r.Sync(bg)
	_, got := files(t, dir)["teammate-new.txt"]
	t.Logf("replica sync with a pending local delete: %v; received teammate file: %v", se, got)
	if e != nil {
		t.Errorf("BUG: deletes are blocked in legacy cross-tenant folders (README: deletes remain legal)")
	}
}
